3. Sum of Digits Until Single Digit
Keep summing the digits of a number until a single-digit result is obtained.
 
public class SumDigit3 {
	static int singleDigit(int n) {
        int sum = 0;
 
        // Repetitively calculate sum until
        // it becomes single digit
        while (n > 0 || sum > 9) {
 
            // If n becomes 0, reset it to sum
            // and start a new iteration.
            if (n == 0) {
                n = sum;
                sum = 0;
            }
 
            sum += n % 10;
            n /= 10;
        }
        return sum;
    }
 
    public static void main(String[] args) {
        int n = 1234;
        System.out.println(singleDigit(n));
    }
 
}

5. Find GCD and LCM of Two Numbers
Use loops and conditionals to compute both GCD and LCM
 
public class GCD_LCM {
 
    public static int findGCD(int no1, int no2) {
        int result = 0;
 
        for (int i = 1; i <= no1; i++) {
            if (no1 % i == 0 && no2 % i == 0)
                result = i;
        }
        return result;
    }
 
    public static int findLCM(int no1, int no2) {
        int result = (no1 * no2) / findGCD(no1, no2);
        return result;
    }
 
    public static void main(String[] args) {
        int no1 = 12;
        int no2 = 18;
 
        System.out.println(findGCD(no1, no2));
        System.out.println(findLCM(no1, no2));
    }
}

    static int getSecondLargest(int[] arr) {
        int n = arr.length;
 
        int largest = -1, secondLargest = -1;
 
        // Finding the largest element
        for (int i = 0; i < n; i++) {
            if (arr[i] > largest)
                largest = arr[i];
        }
 
        // Finding the second largest element
        for (int i = 0; i < n; i++) {
            
            // Update second largest if the current element is greater
            // than second largest and not equal to the largest
            if (arr[i] > secondLargest && arr[i] != largest) {
                secondLargest = arr[i];
            }
        }
        return secondLargest;
    }

9. Find Missing Number in Sequence
Given an array of n-1 numbers from 1 to n, find the missing number.
 
	//The sum of the first n natural numbers is given by the formula (n * (n + 1)) / 2.
//	The idea is to compute this sum and subtract the sum of all elements in the array from it to get the missing number.
	public static void main(String[] args) {
		// TODO Auto-generated method stub
		int[] arr = {8, 2, 4, 5, 3, 7, 1};
        System.out.println(missingNum(arr));
 
	}
	
	public static int missingNum(int[] arr) {
        long n = arr.length + 1;   // Because 1 elemtn is missing as arr.length == n-1 numbers
    
        // Calculate the sum of array elements
        long sum = 0;
        for (int i = 0; i < arr.length; i++) {
            sum += arr[i];
        }
    
        // Use long for expected sum to avoid overflow
        long expSum = n * (n + 1) / 2;
    
        // Return the missing number
        int ans = (int)(expSum - sum);
        
        return ans;
    }

public class ArraySoretdOrNot {
 
    static boolean checkAscending(int[] a) {
 
        for (int i = 0; i < a.length - 1; i++) {
 
            if (a[i] != a[i + 1] && a[i] > a[i + 1])
                return false;
        }
        return true;
    }
 
    static boolean checkDescending(int[] a) {
        for (int i = 0; i < a.length - 1; i++) {
            if (a[i] != a[i + 1] && a[i] < a[i + 1])
                return false;
        }
        return true;
    }
 
    public static void main(String[] args) {
        int[] a = { 1, 1, 2, 3, 4, 5 };

        if (checkAscending(c)) {
            System.out.println("Ascending Array");
        } else if (checkDescending(c)) {
            System.out.println("Descending Array");
        } else {

        }
    }

package temp;

import java.util.Map;

public class LongestWord {
    public static String FrequencyCount(String str) {
        str = str.toLowerCase(); // remove punctuation
        String[] arr = str.split(" ");
        
        HashMap<String, Integer> map = new HashMap<>();

        // store each word and its length
        for (int i = 0; i < arr.length; i++) {
            map.put(arr[i], arr[i].length());
        }

        // variables to track the longest word
        String longestWord = "";
        int maxLength = 0;

        // iterate using forEach
        for (Map.Entry<String, Integer> entry : map.entrySet()) {
            if (entry.getValue() > maxLength) {
                maxLength = entry.getValue();
                longestWord = entry.getKey();
            }
        }

        return longetWord;
    }

16.Armstrong Number Checker
Check if a number is an Armstrong number (e.g., 153 = 1³ + 5³ + 3³).
 
public class AmstrongNumber {
 
    static boolean checkAmstrong(int no) {
        int temp = no;
        int sum = 0;
 
        while (no != 0) {
            int r = no % 10;
//find count digit of number and use it as power
            sum = sum + (r * r * r);
            no = no / 10;
        }
 
        if (temp == sum)
            return true;
        else
            return false;
    }
 
    public static void main(String[] args) {
        if (checkAmstrong(153))
            System.out.println("Amstrong Number");
        else
            System.out.println("Not Amstrong Number");
 
    }
}

22. Find Leader Elements in Array
An element is a leader if it is greater than all elements to its right.
 
// Function to find the leaders in an array
    static ArrayList<Integer> leaders(int[] arr) {
        ArrayList<Integer> result = new ArrayList<>();
        int n = arr.length;
        
        for (int i = 0; i < n; i++) {
            int j;
          
            // Check elements to the right
            for (j = i + 1; j < n; j++) {
              
                // If a larger element is found
                if (arr[i] < arr[j])
                    break;
            }
            
            // If no larger element was found
            if (j == n)
                result.add(arr[i]);
        }
        
        return result;
    }
 
    public static void main(String[] args) {
        int[] arr = { 16, 17, 4, 3, 5, 2 };
        ArrayList<Integer> result = leaders(arr);
        
        for (int res : result) {
            System.out.print(res + " ");
        }
        System.out.println();
    }

23. Matrix Transpose
Transpose a given 2D matrix.
 
public class MatrixTranspose {
 
    public static void main(String[] args) {
        // Original matrix
        int[][] originalMatrix = {
            {1, 2, 3},
            {4, 5, 6},
            {7, 8, 9}
        };
 
        // Get dimensions of the original matrix
        int rows = originalMatrix.length;
        int cols = originalMatrix[0].length;
 
        // Create a new matrix for the transpose with swapped dimensions
        int[][] transposedMatrix = new int[cols][rows];
 
        // Perform the transpose operation
        for (int i = 0; i < rows; i++) {
            for (int j = 0; j < cols; j++) {
                transposedMatrix[j][i] = originalMatrix[i][j];
            }
        }
 
        // Print the original matrix
        System.out.println("Original Matrix:");
        printMatrix(originalMatrix);
 
        // Print the transposed matrix
        System.out.println("\nTransposed Matrix:");
        printMatrix(transposedMatrix);
    }
 
    // Helper method to print a matrix
    public static void printMatrix(int[][] matrix) {
        for (int i = 0; i < matrix.length; i++) {
            for (int j = 0; j < matrix[0].length; j++) {
                System.out.print(matrix[i][j] + " ");
            }
            System.out.println();
        }
    }
}

27. Password Strength Validator
Problem:
Write a program that takes a string input and checks if it qualifies as a strong password. A strong 
password must:
• Be at least 8 characters long
• Contain at least one uppercase letter
• Contain at least one lowercase letter
• Contain at least one digit
• Contain at least one special character (e.g., !@#$%^&*())

package temp;

public class validPassword {

    public static boolean isValidPassword(String str) {
        boolean isUpperCasePresent = false;
        boolean isLowerCasePresent = false;
        boolean isDigitPresent = false;
        boolean isSpecialCharacterPresent = false;

        String specialCharacters = "!@#$%^&*()+";

        if (str.length() < 8)
            return false;

        for (int i = 0; i < str.length(); i++) {
            char ch = str.charAt(i);

            if (Character.isUpperCase(ch))
                isUpperCasePresent = true;
            else if (Character.isLowerCase(ch))
                isLowerCasePresent = true;
            else if (Character.isDigit(ch))
                isDigitPresent = true;
            else if (specialCharacters.contains(ch + ""))   //ch + "" converts char to string
                isSpecialCharacterPresent = true;
        }

        return isUpperCasePresent && isLowerCasePresent
                && isSpecialCharacterPresent && isDigitPresent;
    }

    public static void main(String[] args) {
        System.out.println(isValidPassword("Hello+123")); // true
        System.out.println(isValidPassword("Hello123"));  // false (no special character)
        System.out.println(isValidPassword("hello+123")); // false (no uppercase)
        System.out.println(isValidPassword("HELLO+123")); // false (no lowercase)
        System.out.println(isValidPassword("Hello+12"));  // true
    }
}

public class NumberPyramidPattern {
 
    public static void Pyrimid(int n) {
 
        int no = 1;
 
        for (int i = 1; i <= n; i++) {
 
            for (int j = i; j < n; j++) {
                System.out.print(" ");
            }
 
            for (int k = 1; k <= i; k++) {
                System.out.print(no + " ");
                no++;
            }
            System.out.println();
        }
    }
 
    public static void main(String[] args) {
        int n = 4;
        Pyrimid(n);
    }
}

29.Digit Frequency Counter
Problem:
Take an integer input and count the frequency of each digit (0–9) in it. For example, input 1223334444 
should output:
1 Digit 1: 1 time
2 Digit 2: 2 times
3 Digit 3: 3 times
4 Digit 4: 4 times
 
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class DigitFrequencyCounter {
    public static void main(String[] args) {
        Scanner in = new Scanner(System.in);
        System.out.print("Enter a number: ");
        
        int number = in.nextInt();
        in.close();

        // Convert number to string for easy iteration
        String str = String.valueOf(number);

        // Use HashMap to count frequency of digits
        Map<Character, Integer> frequencyMap = new HashMap<>();

        for (char ch : str.toCharArray()) {
            frequencyMap.put(ch, frequencyMap.getOrDefault(ch, 0) + 1);  //getordefault(ch,0) if get is used first time then default vlue if used after that it will use ch only
        }
//You cAN USE THIS ANOTHER APPROACH FOR LOOP
 //for (char ch : str.toCharArray()) {
 //           Integer count = frequencyMap.get(ch);
 //           if (count == null) {
 //               frequencyMap.put(ch, 1);
 //           } else {
 //              frequencyMap.put(ch, count + 1);
 //            }
 //       }

        // Display digit frequencies
        System.out.println("Digit frequencies:");
        frequencyMap.forEach((digit, count) -> System.out.println(digit + " → " + count));
    }
}
