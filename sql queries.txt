-- List all orders with customer names and product names
 
select orderid , customername , productname from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid;
 
-- Find all customers who placed an order for "Product_14".
select orderid , customername , productname from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
where productname = 'Product_14';
 
-- Show total quantity ordered per customer.
 
select customername , SUM(orders.quantity) as TotalOrders from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
group by customername;
 
-- List customers who have never placed an order.
 
select customername   from customers
LEFT JOIN orders on customers.customerid = orders.customerid
where orders.orderid is null;
 
-- Find products that have never been ordered.
select productname from products
LEFT JOIN orders on products.productid = orders.productid
where orders.orderid is null;
 
-- Show total revenue per product (Price × Quantity).
 
select productname , SUM(Price * Quantity) as Total_Revenue from products
INNER JOIN orders
on products.productid = orders.productid
Group by productname;
 
-- List orders placed in the last 90 days with customer and product details.
 
select orderid , customername , productname from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
where datediff(now(), orderdate) <= 90;
 
-- Find the most frequently ordered product.
 
select productname , COUNT(products.productid) as Total_Orders from products
INNER JOIN orders on products.productid = orders.productid
group by productname
order by Total_Orders DESC
LIMIT 1;
 
-- Show customers who ordered more than 3 different products
 
select customername , count(distinct orders.productid) as TotalOrders from customers
INNER JOIN orders on customers.customerid = orders.customerid
group by customername
having TotalOrders >= 3;
 
-- List cities with total number of orders placed.
 
select city , count(orderid) as Total_Orders from customers
INNER JOIN orders on customers.customerid = orders.customerid
group by city;
 
-- Find customers who ordered products priced above 50.
 
select customername , productname , price from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
where price > 50;
 
-- Show products ordered by customers from "Chicago".
 
select customername , productname , city from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
where city = 'Chicago';
 
-- List customers who ordered the same product more than once.
 
select customername , orders.productid , count(*) as total_orders from customers
INNER JOIN orders on customers.customerid = orders.customerid
group by customername , orders.productid
having total_orders > 1;
 
-- Show the latest order date for each customer.
 
Select customername , MAX(orders.orderdate) as Recent_Order from customers
INNER JOIN orders on customers.customerid = orders.customerid
group by customername;
 
-- Find the top 3 customers by total spending.
 
select customername , SUM(price * quantity) as Total_Spending from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
group by customername
order by Total_Spending desc
LIMIT 3;
 
-- List products ordered by more than 5 different customers.
 
select productname , COUNT(distinct orders.customerid) as Total_customers from orders
INNER JOIN products on products.productid = orders.productid
group by productname
having Total_customers > 5;
 
-- Show average quantity ordered per product.
 
select productname , avg(orders.quantity) as avg_quantity from products
INNER JOIN orders on products.productid = orders.productid
group by productname;
 
-- Find customers who ordered both "Product_1" and "Product_25".
 
select customername from orders
INNER JOIN customers on customers.customerid = orders.customerid
INNER JOIN products on products.productid = orders.productid
where products.productname in ('Product_1' , 'Product_25')
group by customername
having COUNT(distinct products.productname) = 2;
 
-- List all orders with product price and total cost.
 
select orderid , price , (price * quantity) as Total_Cost from products
INNER JOIN orders on products.productid = orders.productid;
 
 
-- Find customers who placed the highest quantity order.
 
select customername from customers
INNER JOIN orders on customers.customerid = orders.customerid
where quantity = (select MAX(quantity) from orders) 
group by customername ;
 
-- List products with price above the average product price.
 
select * from products where price > (select avg(price) from products);
 
-- Find the most expensive product ordered.
 
select productname from products 
INNER JOIN orders on products.productid = orders.productid
where price = (
	select MAX(price) from products
    INNER JOIN Orders on products.productid = orders.productid
);
 
 
-- List customers who ordered the cheapest product.
 
select productname from products 
INNER JOIN orders on products.productid = orders.productid
where price = (
	select MIN(price) from products
    INNER JOIN Orders on products.productid = orders.productid
);
 
-- Show customers whose total spending is above the average spending.
 
with EmplyeeTotalSpending as (
	select 	customername  , SUM(price * quantity) as TotalSpending from orders
	INNER JOIN 	customers on customers.customerid = orders.customerid
	INNER JOIN  products on products.productid = orders.productid
	group by customername
) 
Select customername , TotalSpending from EmplyeeTotalSpending 
where 
TotalSpending > (select avg(TotalSpending) from EmplyeeTotalSpending);
